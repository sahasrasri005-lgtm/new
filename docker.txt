1. docker --version
   Use: Check whether Docker is installed and see its version.

2. docker pull redis
   Use: Download the Redis image from Docker Hub.

3. docker pull ubuntu
   Use: Download the Ubuntu image from Docker Hub.

4. docker images
   Use: List all Docker images available locally.

5. docker image ls
   Use: List all Docker images available locally.

6. docker ps
   Use: Show only currently running containers.

7. docker ps -a
   Use: Show all containers, including stopped containers.

8. docker run --name my-redis -d redis
   Use: Create and run a Redis container in the background.

9. docker run ubuntu
   Use: Run a container from the Ubuntu image.

10. docker run -it ubuntu bash
    Use: Run an Ubuntu container in interactive mode with Bash.

11. docker exec -it my-redis redis-cli
    Use: Open the Redis command-line interface inside the container.

12. SET name "Alice"
    Use: Store the value "Alice" with key "name" in Redis.

13. GET name
    Use: Retrieve the value stored for the key "name".

14. exit
    Use: Exit from Redis CLI or container shell.

15. docker stop my-redis
    Use: Stop a running container.

16. docker start my-redis
    Use: Start a previously stopped container.

17. docker restart my-redis
    Use: Restart a container.

18. docker rm my-redis
    Use: Permanently remove a stopped container.

19. docker rmi redis
    Use: Remove an image from the local system.

20. docker exec -it <container_name> bash
    Use: Open a Bash shell inside a running container.

21. docker build -t redisnew .
    Use: Build a Docker image from the Dockerfile in the current folder.

22. docker run --name myredisnew -d redisnew
    Use: Run a container from your custom-built Redis image.

23. docker login
    Use: Login to your Docker Hub account.

24. docker logout
    Use: Logout from your Docker Hub account.

25. docker commit <container_id> <username>/<image_name>
    Use: Create a new image from the current state of a container.

26. docker tag <image_name> <username>/<image_name>:latest
    Use: Tag an image before pushing it to Docker Hub.

27. docker push <username>/<image_name>:latest
    Use: Upload an image to Docker Hub.

28. docker pull <username>/<image_name>
    Use: Download your image from Docker Hub.

29. docker logs <container_name>
    Use: View the logs generated by a container.

30. docker inspect <container_name>
    Use: View detailed information about a container.

31. docker port <container_name>
    Use: Check the port mapping of a container.

32. docker stats
    Use: Monitor CPU and RAM usage of running containers.

33. docker save -o image.tar <image_name>
    Use: Export a Docker image as a .tar file.

34. docker load -i image.tar
    Use: Load a Docker image from a .tar file.

35. docker run --restart=always <image_name>
    Use: Automatically restart a container if it crashes.

36. docker run -d <image_name>
    Use: Run a container in the background.

37. docker run -p 8080:5000 web_app
    Use: Map host port 8080 to container port 5000.

38. docker build -t lmsimage .
    Use: Build a Docker image for the Maven web application.

39. docker run -d -p 7089:8080 --name lmcontainer lmsimage
    Use: Run the Maven web application and access it on localhost:7089.

40. docker tag lmsimage <username>/lmsimage:latest
    Use: Tag the Maven application image for Docker Hub.

41. docker push <username>/lmsimage:latest
    Use: Push the Maven application image to Docker Hub.


================ DOCKERFILE FOR REDIS ================

FROM redis:latest
Use: Use Redis as the base image.

CMD ["redis-server"]
Use: Start the Redis server when the container starts.


================ DOCKERFILE FOR MAVEN WEB APP ================

FROM tomcat:9.0
Use: Use Tomcat 9 as the base image.

COPY target/*.war /usr/local/tomcat/webapps/ROOT.war
Use: Copy the Maven-generated WAR file into Tomcat.

EXPOSE 7089
Use: Declare the port used by the application.

CMD ["catalina.sh","run"]
Use: Start Tomcat when the container starts.
